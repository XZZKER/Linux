# 链表

节点嵌入宿主结构体，通过 `container_of` 宏获取外层结构。

## API

1. 结构体

    ```c++
    struct list_head {
        struct list_head *next, *prev;
    };
    /*
     * next : 指向后继节点
     * prev : 指向前驱节点
     */
    ```

    

2. 初始化与销毁

    ```c++
    list_head *LIST_HEAD_INIT(name)
    /*
     * name : 待初始化的 list_head 变量名
     * retval : 返回静态初始化器
     */
    
    list_head *LIST_HEAD(name)
    /*
     * name : 要定义的链表头变量名
     * retval : 定义并静态初始化一个链表头
     */
    
    static inline void INIT_LIST_HEAD(struct list_head *list);
    /*
     * list   : 指向待初始化的 list_head 节点
     * retval : 无
     * warning: 动态分配的 list_head 在使用前必须调用此函数初始化
     */
    
    // 销毁：链表自身无资源需释放，只需确保已将所有节点从链表中移除，宿主结构的内存由使用者自行释放。
    ```

    

3. 添加

    ```c++
    void list_add(struct list_head *new, struct list_head *head);
    /*
     * new    : 要插入的新节点
     * head   : 链表头
     * retval : 无
     * warning: 在 head 之后插入
     */
    
    void list_add_tail(struct list_head *new, struct list_head *head);
    /*
     * new    : 要插入的新节点
     * head   : 链表头
     * retval : 无
     * warning: 在 head 之前插入，即链表尾部
     */
    ```

    

4. 删除

    ```c++
    void list_del(struct list_head *entry);
    /*
     * entry  : 要删除的节点
     * retval : 无
     * warning: 仅摘除节点，entry->next/prev 会被设为 POISON 值以捕获悬空引用；
     *          不会释放 entry 或其宿主结构的内存，需自行管理
     */
    
    void list_del_init(struct list_head *entry);
    /*
     * entry  : 要删除的节点
     * retval : 无
     * warning: 与 list_del 相同，但删除后会将 entry 重新初始化为空节点，
     *          可安全再次使用
     */
    ```

    

5. 替换

    ```c++
    void list_replace(struct list_head *old, struct list_head *new);
    /*
     * old    : 要替换的旧节点（必须处于链表中）
     * new    : 新节点
     * retval : 无
     * warning: 替换后 old 的指针处于未定义状态，不初始化；
     *          推荐使用 list_replace_init
     */
    
    void list_replace_init(struct list_head *old, struct list_head *new);
    /*
     * old    : 旧节点
     * new    : 新节点
     * retval : 无
     * warning: 替换后 old 会被初始化为空节点，可再次利用
     */
    ```

    

6. 移动

    ```c++
    void list_move(struct list_head *list, struct list_head *head);
    /*
     * list   : 要移动的节点
     * head   : 目标链表头
     * retval : 无
     * warning: 等价于 list_del + list_add，将节点移动到 head 之后
     */
    
    void list_move_tail(struct list_head *list, struct list_head *head);
    /*
     * list   : 要移动的节点
     * head   : 目标链表头
     * retval : 无
     * warning: 等价于 list_del + list_add_tail，将节点移动到链表尾部
     */
    ```

    

7. 遍历

    ```c++
    list_for_each(pos, head)
    /*
     * pos    : struct list_head 指针，循环变量
     * head   : 链表头指针
     * warning: 遍历过程中不能删除 pos 指向的节点，否则会导致指针失效；
     *          需删除时使用 list_for_each_safe
     */
    
    list_for_each_safe(pos, n, head)
    /*
     * pos    : 当前遍历节点
     * n      : 暂存下一个节点
     * head   : 链表头指针
     * warning: 遍历时可以安全地删除 pos 节点
     */
    
    list_for_each_entry(pos, head, member)
    /*
     * pos    : 宿主结构体类型的指针，循环变量
     * head   : 链表头指针
     * member : list_head 在宿主结构体中的成员名
     * warning: 遍历时不能删除 pos 指向的节点；
     *          需删除时使用 list_for_each_entry_safe
     */
    
    list_for_each_entry_safe(pos, n, head, member)
    /*
     * pos    : 宿主结构体类型的指针，当前遍历节点
     * n      : 同类型指针，暂存下一个节点
     * head   : 链表头指针
     * member : list_head 在宿主结构体中的成员名
     * warning: 遍历时可以安全地删除 pos 节点
     */
    ```

    

8. 辅助

    ```c++
    int list_empty(const struct list_head *head);
    /*
     * head   : 链表头指针
     * retval : 链表为空返回非 0，否则返回 0
     */
    ```

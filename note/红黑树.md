# 红黑树

自平衡二叉搜索树，通过键值比较维护有序性，保证插入、查找、删除操作的时间复杂度为 O(log n)，节点嵌入宿主结构体，通过 `container_of` 宏获取外层结构。

## API

1. 结构体

    ```c++
    struct rb_node {
        unsigned long  __rb_parent_color;
        struct rb_node *rb_right;
        struct rb_node *rb_left;
    };
    /*
     * __rb_parent_color : 低2位存储节点颜色（0=红，1=黑），高位存储父节点指针
     * rb_right          : 右子节点指针
     * rb_left           : 左子节点指针
     */
    
    struct rb_root {
        struct rb_node *rb_node;
    };
    /*
     * rb_node : 指向根节点，空树时为 NULL
     */
    ```

    

2. 初始化与销毁

    ```c++
    /* 动态初始化根节点 */
    static inline void rb_root_init(struct rb_root *root)
    /*
     * root   : 指向要初始化的红黑树根
     * retval : 无
     * warning: 调用前不应有其他引用，必须先初始化才能使用
     */
    ```

    

3. 插入

    ```c++
    void rb_link_node(struct rb_node *node, struct rb_node *parent,
                      struct rb_node **rb_link);
    /*
     * node    : 要插入的新节点（未着色）
     * parent  : 新节点的父节点（若为根可为 NULL）
     * rb_link : 父节点中与新节点连接的指针地址（即 &parent->rb_left 或 &parent->rb_right）
     * retval  : 无
     * warning : 调用前必须通过查找确定插入位置，rb_link 指向该位置指针；
     *           调用后 node 尚未着色，必须紧接着调用 rb_insert_color
     */
    
    void rb_insert_color(struct rb_node *node, struct rb_root *root);
    /*
     * node    : 刚链接好的新节点
     * root    : 红黑树根
     * retval  : 无
     * warning : 必须在 rb_link_node 之后立即调用，完成颜色设置及平衡；
     *           调用期间可能旋转子树，不能并发操作同一棵树
     */
    ```

    

4. 查找

    ```c++
    struct rb_node *rb_first(const struct rb_root *root);
    /*
     * root   : 红黑树根
     * retval : 返回最小键对应的 rb_node，树空返回 NULL
     */
    
    struct rb_node *rb_last(const struct rb_root *root);
    /*
     * root   : 红黑树根
     * retval : 返回最大键对应的 rb_node，树空返回 NULL
     */
    
    struct rb_node *rb_next(const struct rb_node *node);
    /*
     * node   : 当前节点
     * retval : 返回中序后继的 rb_node，若无后继返回 NULL
     */
    
    struct rb_node *rb_prev(const struct rb_node *node);
    /*
     * node   : 当前节点
     * retval : 返回中序前驱的 rb_node，若无前驱返回 NULL
     */
    
    void *rb_entry(ptr, type, member)
    /*
     * ptr    : 指向 struct rb_node 的指针
     * type   : 宿主结构体类型
     * member : rb_node 在宿主结构体中的成员名
     * retval : 返回包含该 rb_node 的宿主结构体指针
     */
    ```

    

5. 删除

    ```c++
    void rb_erase(struct rb_node *node, struct rb_root *root);
    /*
     * node   : 要删除的节点
     * root   : 红黑树根
     * retval : 无
     * warning: 仅从树中移除并重新平衡，不释放 node 或宿主结构体内存；
     *          调用者需自行释放内存，删除后不可再访问该 node
     */
    ```

    

6. 替换

    ```c++
    void rb_replace_node(struct rb_node *victim, struct rb_node *new,
                         struct rb_root *root);
    /*
     * victim : 被替换的节点
     * new    : 新节点（未链接到树中）
     * root   : 红黑树根
     * retval : 无
     * warning: 新节点的左右子节点和颜色将被设为与 victim 相同；
     *          用户需确保新节点的键值与旧节点等效，否则破坏有序性
     */
    ```

    

7. 遍历

    ```c++
    rbtree_postorder_for_each_entry_safe(pos, n, root, member)
    /*
     * pos    : 宿主结构体类型指针，循环变量
     * n      : 同类型指针，暂存下一节点
     * root   : 红黑树根指针
     * member : rb_node 在宿主结构体中的成员名
     * warning: 后序遍历依次访问所有节点，可安全删除节点；
     *          适用于释放整棵树时使用
     */
    ```
